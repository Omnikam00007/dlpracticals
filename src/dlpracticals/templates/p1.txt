# %%
import numpy as np
import matplotlib.pyplot as plt

# %%
def predict(x, w, b):
    return np.where(np.dot(x, w) + b >= 0, 1, 0)


def train(X, y, lr=0.1, epochs=50):
    w = np.zeros(X.shape[1])
    b = 0.0
    for epoch in range(epochs):
        errors = 0
        for xi, target in zip(X, y):
            update = lr * (target - int(predict(xi, w, b)))
            w += update * xi
            b += update
            errors += int(update != 0)
        print(epoch, w, b, errors)
        if errors == 0:
            break
    return w, b

# %%
def plot_boundary(X, y, w, b, title):
    plt.scatter(X[y == 0][:, 0], X[y == 0][:, 1], c='blue', label='class 0')
    plt.scatter(X[y == 1][:, 0], X[y == 1][:, 1], c='red', label='class 1')
    x1 = np.array([X[:, 0].min() - 0.5, X[:, 0].max() + 0.5])
    plt.plot(x1, (-b - w[0] * x1) / w[1], 'k--', label='boundary')
    plt.title(title)
    plt.legend()
    plt.show()

# %%
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
gates = {'AND': np.array([0, 0, 0, 1]), 'OR': np.array([0, 1, 1, 1])}

for name, y in gates.items():
    w, b = train(X, y)
    print(name, 'predictions:', predict(X, w, b), 'target:', y)
    plot_boundary(X, y, w, b, name + ' gate')

# %%
rng = np.random.default_rng(42)
X_syn = np.vstack([rng.normal(-2, 0.8, (100, 2)), rng.normal(2, 0.8, (100, 2))])
y_syn = np.array([0] * 100 + [1] * 100)

w, b = train(X_syn, y_syn, lr=0.01)
print('Accuracy:', (predict(X_syn, w, b) == y_syn).mean())
plot_boundary(X_syn, y_syn, w, b, 'Synthetic data')
