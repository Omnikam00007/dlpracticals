# # Practical 4: Comparing Activation Functions (Sigmoid, Tanh, ReLU)

# %%
import time
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers

# %%
train_df = pd.read_csv(f'{DATA}/mnist_train.csv')
test_df = pd.read_csv(f'{DATA}/mnist_test.csv')

x = train_df.iloc[:, 1:].to_numpy('float32') / 255
y = train_df.iloc[:, 0].to_numpy()
x_test = test_df.iloc[:, 1:].to_numpy('float32') / 255
y_test = test_df.iloc[:, 0].to_numpy()

x_tr, x_val = x[:50000], x[50000:]
y_tr, y_val = y[:50000], y[50000:]

# %%
def build_mlp(activation):
    model = keras.Sequential([keras.Input(shape=(784,))])
    for _ in range(6):
        model.add(layers.Dense(128, activation=activation))
    model.add(layers.Dense(10, activation='softmax'))
    model.compile(optimizer=keras.optimizers.SGD(learning_rate=0.01, momentum=0.9),
                  loss='sparse_categorical_crossentropy', metrics=['accuracy'])
    return model


loss_fn = keras.losses.SparseCategoricalCrossentropy()


def gradient_norms(model):
    with tf.GradientTape() as tape:
        loss = loss_fn(y_tr[:1024], model(x_tr[:1024]))
    grads = tape.gradient(loss, [layer.kernel for layer in model.layers])
    return [float(tf.norm(g)) for g in grads]

# %%
histories, times, grads_before, grads_after, test_acc = {}, {}, {}, {}, {}

for activation in ['sigmoid', 'tanh', 'relu']:
    model = build_mlp(activation)
    grads_before[activation] = gradient_norms(model)
    start = time.time()
    histories[activation] = model.fit(x_tr, y_tr, validation_data=(x_val, y_val),
                                      epochs=15, batch_size=128, verbose=0).history
    times[activation] = time.time() - start
    grads_after[activation] = gradient_norms(model)
    test_acc[activation] = model.evaluate(x_test, y_test, verbose=0)[1]
    print(activation, 'test acc:', round(test_acc[activation], 4), 'time:', round(times[activation], 1), 's')

# %%
fig, axes = plt.subplots(1, 3, figsize=(18, 4))
for activation, h in histories.items():
    axes[0].plot(h['loss'], label=activation)
    axes[1].plot(h['val_accuracy'], label=activation)
axes[0].set_title('Training loss')
axes[1].set_title('Validation accuracy')
axes[2].bar(times.keys(), times.values())
axes[2].set_title('Training time (s)')
for ax in axes[:2]:
    ax.set_xlabel('epoch')
    ax.legend()
plt.show()

# %%
fig, axes = plt.subplots(1, 2, figsize=(14, 4))
for activation in histories:
    axes[0].semilogy(range(1, 8), grads_before[activation], marker='o', label=activation)
    axes[1].semilogy(range(1, 8), grads_after[activation], marker='o', label=activation)
axes[0].set_title('Gradient norm per layer - before training')
axes[1].set_title('Gradient norm per layer - after training')
for ax in axes:
    ax.set_xlabel('layer (1 = input side)')
    ax.legend()
plt.show()

# %%
def epochs_to_95(h):
    hits = [i + 1 for i, acc in enumerate(h['val_accuracy']) if acc >= 0.95]
    return hits[0] if hits else None


pd.DataFrame({
    'test_acc': test_acc,
    'time_s': times,
    'epochs_to_95%': {a: epochs_to_95(h) for a, h in histories.items()},
    'layer1_grad / last_grad': {a: grads_before[a][0] / grads_before[a][-1] for a in histories},
}).round(4)

# %% [markdown]
# ## Vanishing / exploding gradients
# - **Sigmoid**: derivative is at most 0.25, so the gradient shrinks at every layer and the first layers learn very slowly (vanishing gradient).
# - **Tanh**: derivative up to 1 and zero-centred, so it learns faster than sigmoid, but it still saturates for large inputs.
# - **ReLU**: derivative is 1 for positive inputs, so gradients pass through unchanged and training is fastest.
# - **Exploding gradients** happen when each layer multiplies the gradient by more than 1 (large weights, bad initialisation).
# - Fixes: ReLU-family activations, He/Glorot initialisation, batch normalization, residual connections, gradient clipping.
