# %% [markdown]
# # Practical 5: CNN for Image Classification (CIFAR-10)

# %%
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers

# %%
settings = dict(image_size=(32, 32), batch_size=64, seed=42)
train_ds = keras.utils.image_dataset_from_directory(f'{DATA}/cifar10/train', validation_split=0.1,
                                                    subset='training', **settings)
val_ds = keras.utils.image_dataset_from_directory(f'{DATA}/cifar10/train', validation_split=0.1,
                                                  subset='validation', **settings)
test_ds = keras.utils.image_dataset_from_directory(f'{DATA}/cifar10/test', shuffle=False, **settings)
class_names = train_ds.class_names

train_ds = train_ds.cache().shuffle(5000).prefetch(tf.data.AUTOTUNE)
val_ds = val_ds.cache().prefetch(tf.data.AUTOTUNE)
test_ds = test_ds.cache().prefetch(tf.data.AUTOTUNE)

# %%
def build_mlp():
    return keras.Sequential([
        keras.Input(shape=(32, 32, 3)),
        layers.Rescaling(1 / 255),
        layers.Flatten(),
        layers.Dense(512, activation='relu'),
        layers.Dense(256, activation='relu'),
        layers.Dense(10, activation='softmax'),
    ])


def build_cnn(augment=False):
    model = keras.Sequential([keras.Input(shape=(32, 32, 3))])
    if augment:
        model.add(layers.RandomFlip('horizontal'))
        model.add(layers.RandomTranslation(0.1, 0.1))
    model.add(layers.Rescaling(1 / 255))
    model.add(layers.Conv2D(32, 3, padding='same', activation='relu'))
    model.add(layers.MaxPooling2D())
    model.add(layers.Conv2D(64, 3, padding='same', activation='relu'))
    model.add(layers.MaxPooling2D())
    model.add(layers.Flatten())
    model.add(layers.Dense(128, activation='relu'))
    model.add(layers.Dense(10, activation='softmax'))
    return model

# %%
models = {'MLP': build_mlp(), 'CNN': build_cnn(), 'CNN + augmentation': build_cnn(augment=True)}
histories, results = {}, []

for name, model in models.items():
    print('Training', name)
    model.compile(optimizer='adam', loss='sparse_categorical_crossentropy', metrics=['accuracy'])
    histories[name] = model.fit(train_ds, validation_data=val_ds, epochs=15, verbose=2).history
    results.append({'model': name, 'params': model.count_params(),
                    'train_acc': histories[name]['accuracy'][-1],
                    'test_acc': model.evaluate(test_ds, verbose=0)[1]})

# %%
fig, axes = plt.subplots(1, 2, figsize=(14, 4))
for name, h in histories.items():
    axes[0].plot(h['val_accuracy'], label=name)
    axes[1].plot(h['val_loss'], label=name)
axes[0].set_title('Validation accuracy')
axes[1].set_title('Validation loss')
for ax in axes:
    ax.set_xlabel('epoch')
    ax.legend()
plt.show()

pd.DataFrame(results).set_index('model').round(4)

# %%
images, labels = next(iter(test_ds))
predictions = models['CNN + augmentation'].predict(images, verbose=0).argmax(axis=1)

fig, axes = plt.subplots(2, 8, figsize=(16, 5))
for i, ax in enumerate(axes.flat):
    ax.imshow(images[i].numpy().astype('uint8'))
    ax.set_title(f'{class_names[predictions[i]]}\n(true: {class_names[int(labels[i])]})', fontsize=8,
                 color='green' if predictions[i] == int(labels[i]) else 'red')
    ax.axis('off')
plt.show()
