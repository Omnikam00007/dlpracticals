import time
import pandas as pd
import matplotlib.pyplot as plt
from tensorflow import keras
from tensorflow.keras import layers

# %%
train_df = pd.read_csv(f'{DATA}/fashion-mnist_train.csv')
test_df = pd.read_csv(f'{DATA}/fashion-mnist_test.csv')

x = train_df.iloc[:, 1:].to_numpy('float32') / 255
y = train_df.iloc[:, 0].to_numpy()
x_test = test_df.iloc[:, 1:].to_numpy('float32') / 255
y_test = test_df.iloc[:, 0].to_numpy()

x_tr, x_val = x[:50000], x[50000:]
y_tr, y_val = y[:50000], y[50000:]

# %%
def build_model(num_layers, learning_rate):
    model = keras.Sequential([keras.Input(shape=(784,))])
    for i in range(num_layers):
        model.add(layers.Dense(256 if i == 0 else 128, activation='relu'))
        model.add(layers.Dropout(0.2))
    model.add(layers.Dense(10, activation='softmax'))
    model.compile(optimizer=keras.optimizers.Adam(learning_rate),
                  loss='sparse_categorical_crossentropy', metrics=['accuracy'])
    return model

# %%
baseline = {'learning_rate': 1e-3, 'batch_size': 128, 'num_layers': 2}
experiments = {
    'baseline': baseline,
    'lr=0.01': {**baseline, 'learning_rate': 1e-2},
    'lr=0.0001': {**baseline, 'learning_rate': 1e-4},
    'batch=32': {**baseline, 'batch_size': 32},
    'batch=512': {**baseline, 'batch_size': 512},
    'layers=1': {**baseline, 'num_layers': 1},
    'layers=3': {**baseline, 'num_layers': 3},
}

histories, results = {}, []
for name, cfg in experiments.items():
    model = build_model(cfg['num_layers'], cfg['learning_rate'])
    tensorboard = keras.callbacks.TensorBoard(log_dir=f'logs/{name}')
    start = time.time()
    h = model.fit(x_tr, y_tr, validation_data=(x_val, y_val), epochs=10,
                  batch_size=cfg['batch_size'], callbacks=[tensorboard], verbose=0).history
    histories[name] = h
    results.append({'run': name, **cfg,
                    'best_val_acc': max(h['val_accuracy']),
                    'test_acc': model.evaluate(x_test, y_test, verbose=0)[1],
                    'time_s': round(time.time() - start, 1)})
    print(name, 'best val acc:', round(results[-1]['best_val_acc'], 4))

results_df = pd.DataFrame(results).set_index('run')
results_df.round(4)

# %%
%load_ext tensorboard
%tensorboard --logdir logs

# %%
groups = {
    'Learning rate': ['lr=0.01', 'baseline', 'lr=0.0001'],
    'Batch size': ['batch=32', 'baseline', 'batch=512'],
    'Number of layers': ['layers=1', 'baseline', 'layers=3'],
}
fig, axes = plt.subplots(1, 3, figsize=(18, 4))
for ax, (title, names) in zip(axes, groups.items()):
    for name in names:
        ax.plot(histories[name]['val_accuracy'], label=name)
    ax.set_title(title)
    ax.set_xlabel('epoch')
    ax.legend()
plt.show()

# %%
best = results_df['best_val_acc'].idxmax()
print('REPORT')
print('Best configuration:', best, dict(experiments[best]))
print('Best validation accuracy:', round(results_df.loc[best, 'best_val_acc'], 4))
print('Test accuracy:', round(results_df.loc[best, 'test_acc'], 4))
for title, names in groups.items():
    print()
    print(title)
    print(results_df.loc[names, ['best_val_acc', 'test_acc', 'time_s']].round(4).to_string())
